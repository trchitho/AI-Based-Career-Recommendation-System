-- Table: core.roadmap_milestones

-- DROP TABLE IF EXISTS core.roadmap_milestones;

CREATE TABLE IF NOT EXISTS core.roadmap_milestones
(
    id bigint NOT NULL DEFAULT nextval('core.roadmap_milestones_id_seq'::regclass),
    roadmap_id bigint NOT NULL,
    order_no integer,
    skill_name text COLLATE pg_catalog."default",
    description text COLLATE pg_catalog."default",
    estimated_duration text COLLATE pg_catalog."default",
    resources_json jsonb,
    level integer DEFAULT 1,
    CONSTRAINT roadmap_milestones_pkey PRIMARY KEY (id)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.roadmap_milestones
    OWNER to postgres;

COMMENT ON COLUMN core.roadmap_milestones.level
    IS 'Level của milestone (1-6), dùng để phân cấp và giới hạn xem cho free users';
-- Index: idx_milestones_roadmap

-- DROP INDEX IF EXISTS core.idx_milestones_roadmap;

CREATE INDEX IF NOT EXISTS idx_milestones_roadmap
    ON core.roadmap_milestones USING btree
    (roadmap_id ASC NULLS LAST)
    TABLESPACE pg_default;
-- Index: idx_roadmap_milestones_level

-- DROP INDEX IF EXISTS core.idx_roadmap_milestones_level;

CREATE INDEX IF NOT EXISTS idx_roadmap_milestones_level
    ON core.roadmap_milestones USING btree
    (level ASC NULLS LAST)
    TABLESPACE pg_default;
-- Index: idx_roadmap_milestones_roadmap_level

-- DROP INDEX IF EXISTS core.idx_roadmap_milestones_roadmap_level;

CREATE INDEX IF NOT EXISTS idx_roadmap_milestones_roadmap_level
    ON core.roadmap_milestones USING btree
    (roadmap_id ASC NULLS LAST, level ASC NULLS LAST)
    TABLESPACE pg_default;



-- Table: core.roadmaps

-- DROP TABLE IF EXISTS core.roadmaps;

CREATE TABLE IF NOT EXISTS core.roadmaps
(
    id bigint NOT NULL DEFAULT nextval('core.roadmaps_id_seq'::regclass),
    career_id bigint NOT NULL,
    title_en text COLLATE pg_catalog."default",
    title_vn text COLLATE pg_catalog."default",
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now(),
    CONSTRAINT roadmaps_pkey PRIMARY KEY (id),
    CONSTRAINT chk_roadmaps_has_title CHECK (title_en IS NOT NULL OR title_vn IS NOT NULL)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.roadmaps
    OWNER to postgres;
-- Index: idx_roadmaps_title_en

-- DROP INDEX IF EXISTS core.idx_roadmaps_title_en;

CREATE INDEX IF NOT EXISTS idx_roadmaps_title_en
    ON core.roadmaps USING gin
    (to_tsvector('english'::regconfig, title_en))
    TABLESPACE pg_default
    WHERE title_en IS NOT NULL;
-- Index: ux_roadmaps_career_id

-- DROP INDEX IF EXISTS core.ux_roadmaps_career_id;

CREATE UNIQUE INDEX IF NOT EXISTS ux_roadmaps_career_id
    ON core.roadmaps USING btree
    (career_id ASC NULLS LAST)
    TABLESPACE pg_default;

-- Trigger: roadmaps_sync_title

-- DROP TRIGGER IF EXISTS roadmaps_sync_title ON core.roadmaps;

CREATE OR REPLACE TRIGGER roadmaps_sync_title
    BEFORE INSERT OR UPDATE OF title_en, title_vn
    ON core.roadmaps
    FOR EACH ROW
    EXECUTE FUNCTION core.trg_roadmaps_sync_title();




-- Table: core.user_progress

-- DROP TABLE IF EXISTS core.user_progress;

CREATE TABLE IF NOT EXISTS core.user_progress
(
    id bigint NOT NULL DEFAULT nextval('core.user_progress_id_seq'::regclass),
    user_id bigint NOT NULL,
    career_id bigint NOT NULL,
    roadmap_id bigint NOT NULL,
    completed_milestones jsonb DEFAULT '[]'::jsonb,
    milestone_completions jsonb DEFAULT '{}'::jsonb,
    current_milestone_id bigint,
    progress_percentage numeric(5,2) DEFAULT 0,
    started_at timestamp with time zone DEFAULT now(),
    last_updated_at timestamp with time zone DEFAULT now(),
    CONSTRAINT user_progress_pkey PRIMARY KEY (id),
    CONSTRAINT user_progress_user_id_fkey FOREIGN KEY (user_id)
        REFERENCES core.users (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.user_progress
    OWNER to postgres;
-- Index: idx_user_progress_career

-- DROP INDEX IF EXISTS core.idx_user_progress_career;

CREATE INDEX IF NOT EXISTS idx_user_progress_career
    ON core.user_progress USING btree
    (career_id ASC NULLS LAST)
    TABLESPACE pg_default;
-- Index: idx_user_progress_user

-- DROP INDEX IF EXISTS core.idx_user_progress_user;

CREATE INDEX IF NOT EXISTS idx_user_progress_user
    ON core.user_progress USING btree
    (user_id ASC NULLS LAST)
    TABLESPACE pg_default;
