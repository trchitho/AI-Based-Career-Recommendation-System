-- Table: core.course_catalog

-- DROP TABLE IF EXISTS core.course_catalog;

CREATE TABLE IF NOT EXISTS core.course_catalog
(
    id integer NOT NULL DEFAULT nextval('core.course_catalog_id_seq'::regclass),
    external_id character varying(255) COLLATE pg_catalog."default" NOT NULL,
    title character varying(500) COLLATE pg_catalog."default" NOT NULL,
    description text COLLATE pg_catalog."default",
    url character varying(1000) COLLATE pg_catalog."default",
    platform character varying(50) COLLATE pg_catalog."default",
    instructor character varying(255) COLLATE pg_catalog."default",
    rating double precision DEFAULT 0.0,
    num_reviews integer DEFAULT 0,
    price double precision DEFAULT 0.0,
    is_free boolean DEFAULT false,
    level character varying(50) COLLATE pg_catalog."default",
    duration_hrs double precision,
    thumbnail character varying(1000) COLLATE pg_catalog."default",
    language character varying(20) COLLATE pg_catalog."default" DEFAULT 'en'::character varying,
    tags text[] COLLATE pg_catalog."default" DEFAULT '{}'::text[],
    embedding double precision[],
    is_embedded boolean DEFAULT false,
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now(),
    CONSTRAINT course_catalog_pkey PRIMARY KEY (id),
    CONSTRAINT course_catalog_external_id_key UNIQUE (external_id)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.course_catalog
    OWNER to postgres;





-- Table: core.course_skill_map

-- DROP TABLE IF EXISTS core.course_skill_map;

CREATE TABLE IF NOT EXISTS core.course_skill_map
(
    id integer NOT NULL DEFAULT nextval('core.course_skill_map_id_seq'::regclass),
    course_id integer NOT NULL,
    skill_name character varying(255) COLLATE pg_catalog."default" NOT NULL,
    similarity_score double precision NOT NULL,
    created_at timestamp with time zone DEFAULT now(),
    CONSTRAINT course_skill_map_pkey PRIMARY KEY (id),
    CONSTRAINT course_skill_map_course_id_skill_name_key UNIQUE (course_id, skill_name),
    CONSTRAINT course_skill_map_course_id_fkey FOREIGN KEY (course_id)
        REFERENCES core.course_catalog (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.course_skill_map
    OWNER to postgres;
-- Index: ix_csm_skill_score

-- DROP INDEX IF EXISTS core.ix_csm_skill_score;

CREATE INDEX IF NOT EXISTS ix_csm_skill_score
    ON core.course_skill_map USING btree
    (skill_name COLLATE pg_catalog."default" ASC NULLS LAST, similarity_score DESC NULLS FIRST)
    TABLESPACE pg_default;

-- Table: core.roadmaps

-- DROP TABLE IF EXISTS core.roadmaps;

CREATE TABLE IF NOT EXISTS core.roadmaps
(
    id bigint NOT NULL DEFAULT nextval('core.roadmaps_id_seq'::regclass),
    career_id bigint NOT NULL,
    title_en text COLLATE pg_catalog."default",
    title_vn text COLLATE pg_catalog."default",
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now(),
    CONSTRAINT roadmaps_pkey PRIMARY KEY (id),
    CONSTRAINT chk_roadmaps_has_title CHECK (title_en IS NOT NULL OR title_vn IS NOT NULL)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.roadmaps
    OWNER to postgres;
-- Index: idx_roadmaps_title_en

-- DROP INDEX IF EXISTS core.idx_roadmaps_title_en;

CREATE INDEX IF NOT EXISTS idx_roadmaps_title_en
    ON core.roadmaps USING gin
    (to_tsvector('english'::regconfig, title_en))
    TABLESPACE pg_default
    WHERE title_en IS NOT NULL;
-- Index: ux_roadmaps_career_id

-- DROP INDEX IF EXISTS core.ux_roadmaps_career_id;

CREATE UNIQUE INDEX IF NOT EXISTS ux_roadmaps_career_id
    ON core.roadmaps USING btree
    (career_id ASC NULLS LAST)
    TABLESPACE pg_default;

-- Trigger: roadmaps_sync_title

-- DROP TRIGGER IF EXISTS roadmaps_sync_title ON core.roadmaps;

CREATE OR REPLACE TRIGGER roadmaps_sync_title
    BEFORE INSERT OR UPDATE OF title_en, title_vn
    ON core.roadmaps
    FOR EACH ROW
    EXECUTE FUNCTION core.trg_roadmaps_sync_title();
