-- Table: interview.interview_feedback

-- DROP TABLE IF EXISTS interview.interview_feedback;

CREATE TABLE IF NOT EXISTS interview.interview_feedback
(
    id integer NOT NULL DEFAULT nextval('interview.interview_feedback_id_seq'::regclass),
    session_id integer NOT NULL,
    user_id integer NOT NULL,
    question_quality integer,
    ai_accuracy integer,
    overall_experience integer,
    comments text COLLATE pg_catalog."default",
    suggestions text COLLATE pg_catalog."default",
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT interview_feedback_pkey PRIMARY KEY (id),
    CONSTRAINT interview_feedback_session_id_fkey FOREIGN KEY (session_id)
        REFERENCES interview.interview_sessions (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT interview_feedback_user_id_fkey FOREIGN KEY (user_id)
        REFERENCES core.users (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS interview.interview_feedback
    OWNER to postgres;



-- Table: interview.interview_messages

-- DROP TABLE IF EXISTS interview.interview_messages;

CREATE TABLE IF NOT EXISTS interview.interview_messages
(
    id integer NOT NULL DEFAULT nextval('interview.interview_messages_id_seq'::regclass),
    session_id integer NOT NULL,
    role character varying COLLATE pg_catalog."default" NOT NULL,
    content text COLLATE pg_catalog."default" NOT NULL,
    "timestamp" timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    question_type character varying COLLATE pg_catalog."default",
    question_number integer,
    skills_tested jsonb,
    score double precision,
    detailed_scores jsonb,
    feedback text COLLATE pg_catalog."default",
    strengths jsonb,
    weaknesses jsonb,
    suggestion text COLLATE pg_catalog."default",
    has_audio boolean DEFAULT false,
    audio_duration double precision,
    CONSTRAINT interview_messages_pkey PRIMARY KEY (id),
    CONSTRAINT interview_messages_session_id_fkey FOREIGN KEY (session_id)
        REFERENCES interview.interview_sessions (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS interview.interview_messages
    OWNER to postgres;
-- Index: idx_interview_messages_role

-- DROP INDEX IF EXISTS interview.idx_interview_messages_role;

CREATE INDEX IF NOT EXISTS idx_interview_messages_role
    ON interview.interview_messages USING btree
    (role COLLATE pg_catalog."default" ASC NULLS LAST)
    TABLESPACE pg_default;
-- Index: idx_interview_messages_session_id

-- DROP INDEX IF EXISTS interview.idx_interview_messages_session_id;

CREATE INDEX IF NOT EXISTS idx_interview_messages_session_id
    ON interview.interview_messages USING btree
    (session_id ASC NULLS LAST)
    TABLESPACE pg_default;
