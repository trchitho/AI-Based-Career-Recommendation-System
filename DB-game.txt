-- Table: core.assessment_gamification_sessions

-- DROP TABLE IF EXISTS core.assessment_gamification_sessions;

CREATE TABLE IF NOT EXISTS core.assessment_gamification_sessions
(
    id integer NOT NULL DEFAULT nextval('core.assessment_gamification_sessions_id_seq'::regclass),
    assessment_session_id integer NOT NULL,
    user_id integer NOT NULL,
    quiz_mode character varying(50) COLLATE pg_catalog."default" NOT NULL,
    xp_earned integer NOT NULL DEFAULT 0,
    questions_answered integer NOT NULL DEFAULT 0,
    started_at timestamp with time zone NOT NULL DEFAULT now(),
    completed_at timestamp with time zone,
    extra_data jsonb,
    CONSTRAINT assessment_gamification_sessions_pkey PRIMARY KEY (id),
    CONSTRAINT assessment_gamification_sessions_assessment_session_id_fkey FOREIGN KEY (assessment_session_id)
        REFERENCES core.assessment_sessions (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT assessment_gamification_sessions_user_id_fkey FOREIGN KEY (user_id)
        REFERENCES core.users (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.assessment_gamification_sessions
    OWNER to postgres;

COMMENT ON TABLE core.assessment_gamification_sessions
    IS 'Gamification data for each assessment session - does not affect assessment scoring';

COMMENT ON COLUMN core.assessment_gamification_sessions.quiz_mode
    IS 'Quiz mode used: standard, game, or legacy';

COMMENT ON COLUMN core.assessment_gamification_sessions.extra_data
    IS 'Additional gamification data like badges, streaks, etc.';
-- Index: idx_assessment_gamification_sessions_assessment_session_id

-- DROP INDEX IF EXISTS core.idx_assessment_gamification_sessions_assessment_session_id;

CREATE INDEX IF NOT EXISTS idx_assessment_gamification_sessions_assessment_session_id
    ON core.assessment_gamification_sessions USING btree
    (assessment_session_id ASC NULLS LAST)
    TABLESPACE pg_default;
-- Index: idx_assessment_gamification_sessions_user_id

-- DROP INDEX IF EXISTS core.idx_assessment_gamification_sessions_user_id;

CREATE INDEX IF NOT EXISTS idx_assessment_gamification_sessions_user_id
    ON core.assessment_gamification_sessions USING btree
    (user_id ASC NULLS LAST)
    TABLESPACE pg_default;

    achievement_type character varying(100) COLLATE pg_catalog."default" NOT NULL,
    achievement_name character varying(200) COLLATE pg_catalog."default" NOT NULL,
    achievement_description character varying(500) COLLATE pg_catalog."default",
    earned_at timestamp with time zone NOT NULL DEFAULT now(),
    achievement_metadata jsonb,
    CONSTRAINT user_achievements_pkey PRIMARY KEY (id),
    CONSTRAINT user_achievements_user_id_fkey FOREIGN KEY (user_id)
        REFERENCES core.users (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.user_achievements
    OWNER to postgres;
