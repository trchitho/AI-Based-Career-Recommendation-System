-- Table: core.career_group_mapping

-- DROP TABLE IF EXISTS core.career_group_mapping;

CREATE TABLE IF NOT EXISTS core.career_group_mapping
(
    id integer NOT NULL DEFAULT nextval('core.career_group_mapping_id_seq'::regclass),
    career_id bigint NOT NULL,
    group_id integer NOT NULL,
    created_at timestamp with time zone DEFAULT now(),
    CONSTRAINT career_group_mapping_pkey PRIMARY KEY (id),
    CONSTRAINT career_group_mapping_career_id_group_id_key UNIQUE (career_id, group_id),
    CONSTRAINT career_group_mapping_career_id_fkey FOREIGN KEY (career_id)
        REFERENCES core.careers (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT career_group_mapping_group_id_fkey FOREIGN KEY (group_id)
        REFERENCES core.career_groups (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS core.career_group_mapping
    OWNER to postgres;
-- Index: idx_career_group_mapping_career

-- DROP INDEX IF EXISTS core.idx_career_group_mapping_career;

CREATE INDEX IF NOT EXISTS idx_career_group_mapping_career
    ON core.career_group_mapping USING btree
    (career_id ASC NULLS LAST)
    TABLESPACE pg_default;
-- Index: idx_career_group_mapping_group

-- DROP INDEX IF EXISTS core.idx_career_group_mapping_group;

CREATE INDEX IF NOT EXISTS idx_career_group_mapping_group
    ON core.career_group_mapping USING btree
    (group_id ASC NULLS LAST)
    TABLESPACE pg_default;
